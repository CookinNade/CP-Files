#include<bits/stdc++.h>
 
using namespace std;
typedef long long ll;
ll fibonacci(ll n){
    if(n == 0)return 0;
    if(n == 1)return 1;
    return fibonacci(n-1) + fibonacci(n-2);
}
int main () {
    ios_base::sync_with_stdio(false);
    cin.tie(0);
    ll n,x;
    cin>>x>>n;
    cout<<fibonacci(n);
    return 0;
}
